** 基本樣式

org-html 在 Org 9.6 起已自帶 HTML5 與 MathJax 3。這裡只把輸出基調調成「自成一份可讀文檔」：程式碼用 css 配方著色（而非內嵌 inline-CSS）、HTML5 doctype、啟用語意標籤。

  (setq org-html-htmlize-output-type 'css
        org-html-doctype "html5"
        org-html-html5-fancy t))

*** 合成 fancy 樣式

導出的 ~<head>~ 內嵌三樣資產：=misc/org-export-header.html=（字體 preload）、
=misc/org-css/main.js=、=misc/org-css/main.min.css=。CSS/JS 直接塞進導出的 HTML，
自包含、不依賴外部請求。

字體 preload 的 head 片段 =misc/org-export-header.html= 作為資產放在 misc/ 樹下；它
引用與 =main.min.css= 裏 ~@font-face~ 相同的 woff2 檔名（=etbookot-*= 與
=Merriweather-*=）。把導出的 HTML 與 =misc/org-css/=.woff2 放在同一資料夾即可生效，
不需外部 CDN 或手動下載。下列 emacs-lisp 在導出時把它拼進 ~<head>~。

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(with-eval-after-load 'ox-html
(defun ginshio-org-html-asset (name)
  "Return the path to asset NAME under this config's misc/ directory."
  (expand-file-name name (expand-file-name "misc/" ginshio-config-dir)))

(defun ginshio-org-html-reload-fancy-style ()
  "Rebuild `org-html-style-fancy' from the header, JS, and CSS assets."
  (interactive)
  (setq org-html-style-fancy
        (with-temp-buffer
          (insert-file-contents (ginshio-org-html-asset "org-export-header.html"))
          (goto-char (point-max))
          (insert "<script>\n")
          (insert-file-contents (ginshio-org-html-asset "org-css/main.js"))
          (goto-char (point-max))
          (insert "</script>\n<style>\n")
          (insert-file-contents (ginshio-org-html-asset "org-css/main.min.css"))
          (goto-char (point-max))
          (insert "</style>")
          (buffer-string)))
  (when org-fancy-html-export-mode
    (setq org-html-style-default org-html-style-fancy)))

(ginshio-org-html-reload-fancy-style)
)
#+end_src

*** 頁首模板

導出文件的 ~<html>~ 骨架由 ~org-html-template~ 產生。下面以 around advice 改寫它：
在 ~<head>~ 不變的前提下，往 ~<body>~ 開頭塞入亮/暗切換所需的開關結構、並在標題上方
加上日期/作者 meta。

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(defvar org-message-export-in-progress nil
  "Non-nil while an Org to-message export (compat with the old org-msg
setup) is in progress; suppresses the fancy template.")

(defun ginshio-org-html-template-a (orig-fn contents info)
  "Return complete document string after HTML conversion.

CONTENTS is the transcoded contents string.  INFO is a plist holding
export options.  Adds a few extra things to the body compared to the
default implementation."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn contents info)
    (concat
     (when (and (not (org-html-html5-p info)) (org-html-xhtml-p info))
       (let* ((xml-declaration (plist-get info :html-xml-declaration))
              (decl (or (and (stringp xml-declaration) xml-declaration)
                        (cdr (assoc (plist-get info :html-extension)
                                    xml-declaration))
                        (cdr (assoc "html" xml-declaration))
                        "")))
         (when (not (or (not decl) (string= "" decl)))
           (format "%s\n"
                   (format decl
                           (or (and org-html-coding-system
                                    (fboundp 'coding-system-get)
                                    (coding-system-get org-html-coding-system 'mime-charset))
                               "iso-8859-1"))))))
     (org-html-doctype info)
     "\n"
     (concat "<html"
             (cond ((org-html-xhtml-p info)
                    (format
                     " xmlns=\"http://www.w3.org/1999/xhtml\" lang=\"%s\" xml:lang=\"%s\""
                     (plist-get info :language) (plist-get info :language)))
                   ((org-html-html5-p info)
                    (format " lang=\"%s\"" (plist-get info :language))))
             ">\n")
     "<head>\n"
     (org-html--build-meta-info info)
     (org-html--build-head info)
     (org-html--build-mathjax-config info)
     "</head>\n"
     "<body>\n<input type='checkbox' id='theme-switch'><div id='page'>"
     "<label id='switch-label' for='theme-switch'></label>"
     (let ((link-up (org-trim (plist-get info :html-link-up)))
           (link-home (org-trim (plist-get info :html-link-home))))
       (unless (and (string= link-up "") (string= link-home ""))
         (format (plist-get info :html-home/up-format)
                 (or link-up link-home)
                 (or link-home link-up))))
     ;; Preamble.
     (org-html--build-pre/postamble 'preamble info)
     ;; Document contents.
     (let ((div (assq 'content (plist-get info :html-divs))))
       (format "<%s id=\"%s\">\n" (nth 1 div) (nth 2 div)))
     ;; Document title.
     (when (plist-get info :with-title)
       (let ((title (and (plist-get info :with-title)
                         (plist-get info :title)))
             (subtitle (plist-get info :subtitle))
             (html5-fancy (org-html--html5-fancy-p info)))
         (when title
           (format
            (if html5-fancy
                "<header class=\"page-header\">%s\n<h1 class=\"title\">%s</h1>\n%s</header>"
              "<h1 class=\"title\">%s%s</h1>\n")
            (if (or (plist-get info :with-date)
                    (plist-get info :with-author))
                (concat "<div class=\"page-meta\">"
                        (when (plist-get info :with-date)
                          (org-export-data (plist-get info :date) info))
                        (when (and (plist-get info :with-date) (plist-get info :with-author)) ", ")
                        (when (plist-get info :with-author)
                          (org-export-data (plist-get info :author) info))
                        "</div>\n")
              "")
            (org-export-data title info)
            (if subtitle
                (format
                 (if html5-fancy
                     "<p class=\"subtitle\" role=\"doc-subtitle\">%s</p>\n"
                   (concat "\n" (org-html-close-tag "br" nil info) "\n"
                           "<span class=\"subtitle\">%s</span>\n"))
                 (org-export-data subtitle info))
              "")))))
     contents
     (format "</%s>\n" (nth 1 (assq 'content (plist-get info :html-divs))))
     ;; Postamble.
     (org-html--build-pre/postamble 'postamble info)
     ;; Closing document.
     "</div>\n</body>\n</html>")))

(advice-add #'org-html-template :around #'ginshio-org-html-template-a)
#+end_src

*** 可摺疊的 src 與示例塊

GitHub 式摺疊：把 ~<pre>~ 包在 ~<details>~ 裏，不需 CSS 就有原生可摺疊。預設全部
展開（`open`）。要讓某塊預設收合，加 ~#+attr_html: :collapsed t~；整份文件用
~#+OPTIONS: collapsed:t~。

首先要讓 ox-html 認識 :collapsed 選項。

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(defvar ginshio-org-html-collapsed nil
  "Whether src blocks collapse by default in exports.")

(with-eval-after-load 'ox-html
  ;; `eval' defers macro expansion to run time: `cl-pushnew' on the struct
  ;; accessor would otherwise be expanded (and compiled to a call to
  ;; `(setf org-export-backend-options)') before `ox' defines the struct.
  (eval '(cl-pushnew '(:collapsed "COLLAPSED" "collapsed"
                                  ginshio-org-html-collapsed t)
                     (org-export-backend-options
                      (org-export-get-backend 'html))))
  (add-to-list 'org-default-properties "EXPORT_COLLAPSED"))
#+end_src

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(defvar ginshio-org-html-lang-name-alist
  '(("emacs-lisp" . "Emacs Lisp") ("elisp" . "Emacs Lisp")
    ("cpp" . "C++") ("C" . "C")
    ("js" . "Javascript") ("sh" . "Shell Script") ("bash" . "Bash")
    ("plain-tex" . "TeX") ("tex" . "LaTeX")
    ("conf" . "Configuration File"))
  "Alist of (MODE-NAME . DISPLAY-NAME) for source block language names.")

(defun ginshio-org-html-lang-name (mode)
  "Convert MODE string to a human-readable language name.
First checks `ginshio-org-html-lang-name-alist', then falls back to the
capitalized MODE.  Returns the empty string when MODE is nil (a src block
with no :language), never errors."
  (or (cdr (assoc mode ginshio-org-html-lang-name-alist))
      (if mode (capitalize mode) "")))

(defun ginshio-org-html-src-block-a (orig-fn src-block contents info)
  "Wrap the usual <pre> block in a <details> display."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn src-block contents info)
    (let* ((properties (cadr src-block))
           (lang (ginshio-org-html-lang-name
                  (plist-get properties :language)))
           (name (plist-get properties :name))
           (ref (org-export-get-reference src-block info))
           (collapsed-p (member (or (org-export-read-attribute :attr_html src-block :collapsed)
                                    (plist-get info :collapsed))
                                '("y" "yes" "t" t "true" "all"))))
      (format
       "<details id='%s' class='code'%s><summary%s>%s</summary>
<div class='gutter'>
<a href='#%s'>#</a>
<button title='Copy to clipboard' onclick='copyPreToClipbord(this)'>⎘</button>\
</div>
%s
</details>"
       ref
       (if collapsed-p "" " open")
       (if name " class='named'" "")
       (concat
        (when name (concat "<span class=\"name\">" name "</span>"))
        "<span class=\"lang\">" lang "</span>")
       ref
       (if name
           (replace-regexp-in-string (format "<pre\\( class=\"[^\"]+\"\\)? id=\"%s\">" ref) "<pre\\1>"
                                     (funcall orig-fn src-block contents info))
         (funcall orig-fn src-block contents info))))))

(advice-add #'org-html-src-block :around #'ginshio-org-html-src-block-a)
#+end_src

example 塊、fixed-width（= 逐字段）與 property drawer 也用同樣的 ~<details>~ 摺疊。
property drawer 預設收合（屬性表通常只是包袱）。

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(defun ginshio-org-html-block-a (orig-fn block contents info)
  "Wrap the usual example/fixed-width/property block in a <details>."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn block contents info)
    (let* ((ref (org-export-get-reference block info))
           (type (pcase (car block)
                   ('property-drawer "Properties")))
           (collapsed-default (pcase (car block)
                                ('property-drawer t)
                                (_ nil)))
           (rendered (funcall orig-fn block contents info))
           ;; The <details> wrapper already carries the anchor id; drop the
           ;; duplicate one org puts on the inner <pre> for labelled blocks.
           (id-stripped (replace-regexp-in-string
                         (format "<pre\\( class=\"[^\"]+\"\\)? id=\"%s\">" ref)
                         "<pre\\1>" rendered))
           (collapsed-p (or (member (org-export-read-attribute :attr_html block :collapsed)
                                    '("y" "yes" "t" t "true"))
                            (member (plist-get info :collapsed) '("all")))))
      (format
       "<details id='%s' class='code'%s>
<summary%s>%s</summary>
<div class='gutter'>\
<a href='#%s'>#</a>
<button title='Copy to clipboard' onclick='copyPreToClipbord(this)'>⎘</button>\
</div>
%s\n
</details>"
       ref
       (if (or collapsed-p collapsed-default) "" " open")
       (if type " class='named'" "")
       (if type (format "<span class='type'>%s</span>" type) "")
       ref
       id-stripped))))

(advice-add #'org-html-example-block   :around #'ginshio-org-html-block-a)
(advice-add #'org-html-fixed-width     :around #'ginshio-org-html-block-a)
(advice-add #'org-html-property-drawer :around #'ginshio-org-html-block-a)
#+end_src

要處理寬表格——尤其行動裝置上——需要把 overflow 控制在容器。~table~ 元素不能直接
設 max-width/overflow，所以包一層 ~div~，並像 src 塊一樣加連結錨與 ~#+name~（若有的話）。

(defun ginshio-org-html-table-a (orig-fn table contents info)
  "Wrap the usual <table> in a <div>."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn table contents info)
    (let* ((name (plist-get (cadr table) :name))
           (ref (org-export-get-reference table info)))
      (format "<div id='%s' class='table'>
<div class='gutter'><a href='#%s'>#</a></div>
<div class='tabular'>
%s
</div>\
</div>"
              ref ref
              (if name

                                            (funcall orig-fn table contents info))

*** 可展開的目錄樹

純 CSS 無法做出可收起/展開的 TOC 樹，但可以借 ~<details>~ 或 checkbox＋label 達到。
舊配置選擇用 checkbox＋label、配合 ~org-html--toc-text~ 的正則把「葉」顆粒的 label
拆掉，做出樹狀折疊。照搬。

#+begin_src emacs-lisp :tangle "lisp/ginshio/ginshio-org-html.el"
(defun ginshio-org-html-toc-headline-a (orig-fn headline info)
  "Add a label and checkbox to `org-html--format-toc-headline's output,
making the TOC a collapsible tree."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn headline info)
    (let ((id (or (org-element-property :CUSTOM_ID headline)
                  (org-export-get-reference headline info))))
      (format "<input type='checkbox' id='toc--%s'/><label for='toc--%s'>%s</label>"
              id id (funcall orig-fn headline info)))))

(advice-add #'org-html--format-toc-headline
            :around #'ginshio-org-html-toc-headline-a)

(defun ginshio-org-html-toc-text-a (orig-fn toc-entries &optional scope)
  "Remove label for leaf headings in the TOC text."
  (if (or (not org-fancy-html-export-mode)
          (bound-and-true-p org-message-export-in-progress))
      (funcall orig-fn toc-entries scope)
    (replace-regexp-in-string
     "<input [^>]+><label [^>]+>\\(.+?\\)</label></li>"
     "\\1</li>"
     (funcall orig-fn toc-entries scope))))

(advice-add #'org-html--toc-text :around #'ginshio-org-html-toc-text-a)
#+end_src
